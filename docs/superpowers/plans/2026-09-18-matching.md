```js
/**
 * Is this candidate learning the same language as the viewer?
 *
 * The "study buddy" relationship, which the scorer had no branch for at all.
 * Measured over 2,800 production conversation pairs: study buddies reply at
 * 16.2%, above the one-way tier's 12.6% -- while scoring zero and then being
 * removed by the languageScore > 0 filter.
 *
 * Uses the same lowercased equivalence set as every other predicate here, so
 * "English", "English (US)" and "en" are one target. An empty value yields an
 * empty set and `$in []` is always false, so a viewer with no target language
 * matches nobody rather than everybody.
 */
const sameTargetExpression = (viewer = {}) => ({
  $in: [
    loweredField('$language_to_learn'),
    { $literal: equivalentValues(viewer.language_to_learn) },
  ],
});
```

and add `sameTargetExpression` to `module.exports`.

```js
'use strict';

const {
  languageMatchExpressions,
  perfectPairExpression,
  sameTargetExpression,
} = require('./matchLanguage');

/**
 * Scoring weights, shared by /matching/recommendations and the ranked
 * /users path so the two cannot drift.
 *
 * The language tiers are ordered by MEASURED reply rate over 2,800 production
 * conversation pairs (2026-09-18): perfect 18.7%, study buddy 16.2%, one-way
 * 12.6%. Before this, both partial relationships paid a flat 20 and the
 * study-buddy relationship had no branch at all -- so the second-best tier
 * scored zero while the worst scored 20.
 *
 * languageMatch (50) stays dominant: everything else still sums to well under
 * it, so a perfect-language partner with no bonuses beats a partial maxed on
 * extras.
 */
const WEIGHTS = {
  languageMatch: 50,
  sameTarget: 24,
  languagePartial: 16,
  activeRecently: 15,
  activeThisWeek: 10,
  sameCity: 8,
  sameCountry: 4,
  levelComplement: 6,
  profileComplete: 5,
  intentOverlap: 10,
  recentlyShown: -10,
  // Spec 3.3. Zero subscribers exist today (0 of 1,856), so this is inert on
  // arrival -- which is the point. The legacy sort makes VIP the PRIMARY key,
  // so the day someone finally subscribes it would reorder the whole screen
  // above language. As a small additive bonus it nudges instead.
  vip: 5,
};

/**
 * Branch order is precedence, and it matters: a perfect pair is ALSO a
 * one-way match by definition, so perfect must be tested first. Same-target
 * sits second because it outranks one-way.
 */
const languageScoreExpression = (viewer) => {
  const { theirNativeIsMyLearning, theirLearningIsMyNative } =
    languageMatchExpressions(viewer);
  return {
    $switch: {
      branches: [
        { case: perfectPairExpression(viewer), then: WEIGHTS.languageMatch },
        { case: sameTargetExpression(viewer), then: WEIGHTS.sameTarget },
        { case: theirNativeIsMyLearning, then: WEIGHTS.languagePartial },
        { case: theirLearningIsMyNative, then: WEIGHTS.languagePartial },
      ],
      default: 0,
    },
  };
};

/**
 * Every scoring stage, for one viewer.
 *
 * `randomJitter` is a caller-supplied NUMBER, not $rand: on a paginated path a
 * per-request random re-rolls between pages, which shows some candidates twice
 * and hides others entirely. Callers that do not paginate may pass a $rand
 * expression; callers that do must pass a per-viewer-per-day stable seed or 0.
 */
const scoringStages = ({ viewer, seenIds = [], viewerIntents = [], randomJitter = 0 }) => {
  const now = Date.now();
  const { theirLearningIsMyNative } = languageMatchExpressions(viewer);

  return [
    {
      $addFields: {
        languageScore: languageScoreExpression(viewer),
        activityScore: {
          $switch: {
            branches: [
              { case: { $gte: ['$lastActive', new Date(now - 24 * 60 * 60 * 1000)] },
                then: WEIGHTS.activeRecently },
              { case: { $gte: ['$lastActive', new Date(now - 7 * 24 * 60 * 60 * 1000)] },
                then: WEIGHTS.activeThisWeek },
            ],
            default: 0,
          },
        },
        profileScore: {
          $add: [
            { $cond: [{ $gte: [{ $size: { $ifNull: ['$images', []] } }, 2] }, 3, 0] },
            { $cond: [{ $gt: [{ $strLenCP: { $ifNull: ['$bio', ''] } }, 20] }, 2, 0] },
          ],
        },
        locationScore: {
          $switch: {
            branches: [
              { case: { $and: [
                  { $ne: ['$location.city', null] },
                  { $ne: ['$location.city', ''] },
                  { $eq: ['$location.city', viewer.location?.city || null] },
                ] }, then: WEIGHTS.sameCity },
              { case: { $and: [
                  { $ne: ['$location.country', null] },
                  { $ne: ['$location.country', ''] },
                  { $eq: ['$location.country', viewer.location?.country || null] },
                ] }, then: WEIGHTS.sameCountry },
            ],
            default: 0,
          },
        },
        levelComplementScore: {
          $cond: [
            { $and: [
                theirLearningIsMyNative,
                { $in: ['$languageLevel', ['B1', 'B2', 'C1', 'C2']] },
              ] },
            WEIGHTS.levelComplement,
            0,
          ],
        },
        intentScore: {
          $cond: [
            { $gt: [
                { $size: { $setIntersection: [
                    { $ifNull: ['$intents', []] }, viewerIntents,
                  ] } },
                0,
              ] },
            WEIGHTS.intentOverlap,
            0,
          ],
        },
        seenPenalty: {
          $cond: [{ $in: ['$_id', seenIds] }, WEIGHTS.recentlyShown, 0],
        },
        vipScore: {
          $cond: [{ $eq: ['$vipSubscription.isActive', true] }, WEIGHTS.vip, 0],
        },
        randomFactor: randomJitter,
      },
    },
    {
      $addFields: {
        matchScore: {
          $add: [
            '$languageScore', '$activityScore', '$profileScore', '$locationScore',
            '$levelComplementScore', '$intentScore', '$seenPenalty', '$vipScore',
            '$randomFactor',
          ],
        },
      },
    },
  ];
};

module.exports = { WEIGHTS, languageScoreExpression, scoringStages };
```
